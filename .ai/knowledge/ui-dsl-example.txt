@overlay.recording {
  extends: overlay.idle
  size: 80x24
  shape: pill(radius:12)
  fill: rgba(76,175,140,0.95)  // sage green
  shadow: 0 6 20 rgba(76,175,140,0.4)  // green glow

  transition: from(overlay.idle) {
    duration: 200ms
    easing: ease-out
    props: [width, height, background]
  }

  children {
    waveform {
      bars: 12
      bar-width: 2.5
      spacing: 1
      color: white
      height: 3..16  // min..max
      animation: sine(freq:10hz, fps:20)
    }
  }
}

  background: transparent
  color: #4CAF8C

// Conditional rendering
@model-card {
  when: state.installed {
    badge("✓ Installed") {
      color: #4CAF8C
      position: top-right
    }
    button("Set Active") {
      disabled: state.active
    }
  }

  when: state.downloading {
    progress-bar {
      value: state.progress
      color: #4CAF8C
    }
    text("${state.downloaded} / ${state.total}") {
      font: 11/regular
      color: #666666
    }
  }
}
